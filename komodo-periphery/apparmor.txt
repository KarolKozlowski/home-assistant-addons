#include <tunables/global>

profile komodo-periphery flags=(attach_disconnected,mediate_deleted) {
  #include <abstractions/base>

  # Capabilities
  file,
  capability sys_admin,
  capability mount,
  signal (send) set=(kill,term,int,hup,cont),

  # Allow S6 cont-init.d to mount cgroup2 v2 hierarchy if needed
  mount fstype=cgroup2 -> /sys/fs/cgroup/**,

  # S6-Overlay
  /init ix,
  /bin/** ix,
  /usr/bin/** ix,
  /run/{s6,s6-rc*,service}/** ix,
  /package/** ix,
  /command/** ix,
  /etc/services.d/** rwix,
  /etc/cont-init.d/** rwix,
  /etc/cont-finish.d/** rwix,
  /run/{,**} rwk,
  /dev/tty rw,

  # Bashio
  /usr/lib/bashio/** ix,
  /tmp/** rwk,

  # Access to options.json and other files within your app
  /data/** rw,

  # Start new profile for service
  /usr/local/bin/periphery cx -> periphery,

  profile periphery flags=(attach_disconnected,mediate_deleted) {
    #include <abstractions/base>
    #include <abstractions/nameservice>

    # Receive signals from S6-Overlay
    signal (receive) peer=komodo-periphery,

    # Periphery connects to Komodo Core and the Docker API
    network,
    /var/run/docker.sock rw,

    # Capabilities for sysinfo / container metrics
    capability sys_ptrace,
    capability sys_admin,

    # Access to options.json and data directory
    /data/** rwix,

    # --- Container & System Metrics Access ---
    # cgroup v1 & v2 paths
    /sys/fs/cgroup/** r,
    /sys/fs/cgroup/ r,
    /sys/devices/system/cpu/** r,
    /sys/devices/system/node/** r,

    # Disk & Proc statistics resolution
    /etc/hostname r,
    /proc/mounts r,
    /proc/stat r,
    /proc/meminfo r,
    /proc/uptime r,
    /proc/sys/fs/binfmt_misc/ r,
    /proc/sys/kernel/osrelease r,
    /proc/[0-9]*/{stat,statm,status,cgroup,cmdline,io} r,
    # ----------------------------------------

    # Access required for service functionality
    /usr/local/bin/periphery r,
    /usr/bin/** ix,
    /usr/local/bin/** ix,
    /bin/** ix,
    /bin/bash rix,
    /bin/echo ix,
    /run/** rwk,
    /tmp/** rwkix,
    /dev/** rw,
    /proc/** r,
    /sys/** r,
    /etc/ssl/** r,
    /etc/ca-certificates/** r,
    /usr/share/ca-certificates/** r,
    /etc/profile r,
    /etc/profile.d/** r,
    /etc/bash.bashrc r,
    /etc/inputrc r,
    /root/ rix,
    /root/**/ rix,
    /root/** r,
    /root/.bash_history rw,
    /root/.cache/ rwix,
    /root/.cache/starship/ rwix,
    /root/.cache/starship/** rwk,
    /etc/passwd r,
    /etc/group r,
    /dev/tty rw,
  }
}
